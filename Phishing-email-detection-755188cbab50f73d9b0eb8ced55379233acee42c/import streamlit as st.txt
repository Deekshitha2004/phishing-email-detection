def extract_urls(text):
    """Extract URLs from email text"""
    url_pattern = r'http[s]?://(?:[a-zA-Z]|[0-9]|[$-_@.&+]|[!*\\(\\),]|(?:%[0-9a-fA-F][0-9a-fA-F]))+'
    urls = re.findall(url_pattern, text)
    return urls
def extract_suspicious_keywords(text):
    """Extract suspicious keywords commonly found in phishing emails"""
    suspicious_words = [
        'verify', 'confirm', 'update', 'click here', 'urgent', 'act now',
        'expire', 'suspended', 'limited time', 'claim', 'reset password',
        'validate', 'authenticate', 'confirm identity', 'unusual activity',
        'account locked', 'take action'
    ]
    found_keywords = []
    text_lower = text.lower()
    # Use word-boundary matching to reduce false positives from substrings
    for word in suspicious_words:
        pattern = r"\b" + re.escape(word) + r"\b"
        if re.search(pattern, text_lower):
            found_keywords.append(word)
    return found_keywords
def check_sender_domain(email_text):
    """Extract and check sender domain (case-insensitive)"""
    from_pattern = r'from:\s*([^\n]+@[^\s]+)'
    matches = re.findall(from_pattern, email_text, flags=re.IGNORECASE)
    return matches
def get_confidence_score(prediction=None, prediction_proba=None):
    """Get model confidence score (helper)"""
    if prediction_proba is not None:
        try:
            return max(prediction_proba) * 100
        except:
            return 50.0
    return 50.0
def extract_advanced_features(email_text):
    """Extract advanced features for improved accuracy"""
    features = {
        'has_misspelled_domains': 0,
        'suspicious_spelling_count': 0,
        'has_html_tags': 0,
        'excessive_capitalization': 0,
        'has_attachments_mention': 0,
        'has_urgent_language': 0,
        'url_domain_mismatch': 0,
    }
    text_lower = email_text.lower()
    # Check for common misspelled domains (homograph attacks)
    misspelled_domains = ['paypa.l', 'amazo.n', 'microso.ft', 'app1e', 'g00gle']
    for domain in misspelled_domains:
        if domain in text_lower:
            features['has_misspelled_domains'] += 1
    # Check for suspicious misspellings
    misspellings = ['veify', 'confim', 'urgnet', 'imidiate', 'immediat']
    for word in misspellings:
        if word in text_lower:
            features['suspicious_spelling_count'] += 1
    # Check for HTML tags (spam/phishing often use HTML)
    if '<a ' in email_text or '<form' in email_text or 'onclick' in text_lower:
        features['has_html_tags'] += 1
    # Check for excessive capitalization (spam indicator)
    caps_ratio = sum(1 for c in email_text if c.isupper()) / max(len(email_text), 1)
    if caps_ratio > 0.3:
        features['excessive_capitalization'] = 1
    # Check for attachments mention
    if any(word in text_lower for word in ['attached', 'attachment', 'download file', 'open attachment']):
        features['has_attachments_mention'] = 
    # Check for urgent/threatening language patterns
    urgent_patterns = [
        'confirm within', 'verify within', 'action required within',
        'will be locked', 'will be closed', 'must act', 'do not ignore',
        'failure to', 'compromise', 'security threat'
    ]
    for pattern in urgent_patterns:
        if pattern in text_lower:
            features['has_urgent_language'] += 1
    return features
def calculate_risk_score(email_text):
    """Calculate risk score based on heuristics"""
    risk_score = 0
    risk_factors = []
    text_lower = email_text.lower()
    # Advanced features analysis
    advanced_features = extract_advanced_features(email_text)
    if advanced_features['has_misspelled_domains']:
        risk_score += 40
        risk_factors.append(f"🚨 Misspelled domain detected - likely homograph attack")
    if advanced_features['suspicious_spelling_count']:
        risk_score += 25
        risk_factors.append(f"⚠️ Suspicious misspellings detected ({advanced_features['suspicious_spelling_count']})")
    if advanced_features['has_html_tags']:
        risk_score += 15
        risk_factors.append(f"📄 HTML/Form tags detected in email body")
    
    if advanced_features['excessive_capitalization']:
        risk_score += 10
        risk_factors.append(f"📢 Excessive capitalization detected")
    if advanced_features['has_attachments_mention']:
        risk_score += 20
        risk_factors.append(f"📎 Suspicious attachment reference")
    if advanced_features['has_urgent_language']:
        risk_score += advanced_features['has_urgent_language'] * 15
        risk_factors.append(f"🚨 Urgent/threatening language detected ({advanced_features['has_urgent_language']} instances)")
    # CRITICAL PHISHING INDICATORS (immediate red flags)
    critical_indicators = {
        'verify account': 30,
        'confirm identity': 30,
        'reset password': 35,
        'click here': 25,
    }
    for indicator, points in critical_indicators.items():
        if indicator in text_lower:
            risk_score += points
            risk_factors.append(f"⚠️ CRITICAL: '{indicator}' detected")
    ]
    keyword_matches = 0
    for kw in suspicious_keywords:
        if re.search(r"\b" + re.escape(kw) + r"\b", text_lower):
            keyword_matches += 1
    if keyword_matches > 0:
        # SIGNIFICANTLY REDUCE per-keyword weight to avoid false positives
        keyword_points = min(20, keyword_matches * 3)  # Reduced from 6 to 3, max 20 instead of 40
        risk_score += keyword_points
        risk_factors.append(f"🚩 Business keywords: {keyword_matches} found ({keyword_points} pts)")
    # PHISHING PATTERNS - URLS (REDUCED WEIGHT for legitimate emails with links)
    urls = extract_urls(email_text)
    if urls:
        # Only penalize if many URLs or suspicious shorteners
        url_points = min(15, len(urls) * 5)  # Reduced from 15 per URL
        risk_score += url_points
        risk_factors.append(f"🔗 URLs detected: {len(urls)} ({url_points} pts)")
        # Check for suspicious URL patterns
        for url in urls:
            if any(x in url.lower() for x in ['bit.ly', 'tinyurl', 'short.url', 'bit.', 'url.']):
                risk_score += 20
                risk_factors.append(f"⚠️ SHORTENED URL: {url}")
    # GENERIC GREETINGS (REDUCED - common in legitimate newsletters)
    generic_greetings = ['dear customer', 'dear user', 'hello user', 'dear member', 'dear sir', 'dear madam', 'valued customer']
    for greeting in generic_greetings:
        if greeting in text_lower:
            risk_score += 5  # Reduced from 25
            risk_factors.append(f"👤 Generic greeting: '{greeting}'")
            break
    # URGENCY/TIME PRESSURE (REDUCED - common in legitimate business)
    urgency_words = ['urgent', 'immediately', 'asap', '24 hour', '48 hour', 'expire', 'expired', 'ending soon', 'limited time']
    urgency_count = 0
    for word in urgency_words:
        if word in text_lower:
            urgency_count += 1
    if urgency_count > 0:
        risk_score += min(10, urgency_count * 3)  # Reduced significantly from 20 each
        risk_factors.append(f"⏰ Urgency language: {urgency_count} found")
    # REQUESTING SENSITIVE INFO (keep high - this IS a strong phishing indicator)
    sensitive_info = ['password', 'credit card', 'social security', 'ssn', 'pin', 'cvv', 'bank account', 'login credential']
    sensitive_count = 0
    for info in sensitive_info:
        if info in text_lower:
            sensitive_count += 1
    if sensitive_count > 0:
        risk_score += sensitive_count * 25  # Keep this high
        risk_factors.append(f"🔐 Requests sensitive info: {sensitive_count} type(s)")
    # SENDER DOMAIN ISSUES (check allowlist and suspicious patterns)
    senders = check_sender_domain(email_text)
    if senders:
        for sender in senders:
            if '@' in sender:
                domain = sender.split('@')[-1].lower()
                local = sender.split('@')[0].lower()
                # allowlist check: if sender or domain is trusted, reduce risk
                if domain in st.session_state.allowlist_domains or sender.lower() in st.session_state.allowlist_senders or local in st.session_state.allowlist_senders:
                    risk_factors.append(f"✅ Sender/domain whitelisted: {sender}")
                    continue
                # Common spoofed domains
                if any(x in domain for x in ['paypa.', 'amazn', 'googl', 'microsoft.', 'apple.', '.tk', '.ml']):
                    risk_score += 35
                    risk_factors.append(f"⚠️ SUSPICIOUS DOMAIN: {domain}")
    # LENGTH CHECK - Very short emails are often phishing
    if len(email_text) < 100:
        risk_score += 15
        risk_factors.append("📏 Email is unusually short")
    if not any(word in text_lower for word in ['mr.', 'mrs.', 'ms.', 'dr.', 'name', 'account number']):
        risk_score += 10
        risk_factors.append("📭 No personalization detected")
    return min(100, risk_score), risk_factors

        if result['suspicious_keywords']:
            pdf.multi_cell(0, 5, f"Suspicious Keywords: {', '.join(result['suspicious_keywords'][:5])}")

def generate_csv_report(results):
    """Generate CSV report of analysis"""
    output = io.StringIO()
    writer = csv.writer(output)
    writer.writerow(['Timestamp', 'Status', 'Confidence (%)', 'URLs Found', 'Suspicious Keywords', 'Preview'])   
    for result in results:
        writer.writerow([
            result['timestamp'],
            'PHISHING' if result['is_phishing'] else 'SAFE',
            f"{result['confidence']:.2f}",
            len(result['urls']),
            len(result['suspicious_keywords']),
            result['email_preview']
        ])    
    return output.getvalue().encode('utf-8')

def _decode_mime_words(s):
    try:
        parts = decode_header(s or '')
    except Exception:
        return s or ''
    decoded = ''
    for part, enc in parts:
        if isinstance(part, bytes):
            try:
                decoded += part.decode(enc or 'utf-8', errors='ignore')
            except Exception:
                decoded += part.decode('utf-8', errors='ignore')
        else:
            decoded += part
    return decoded
def fetch_unread_emails(imap_server, username, password, mailbox='INBOX', limit=10, mark_seen=True):
    results = []
    try:
        imap = imaplib.IMAP4_SSL(imap_server)
        imap.login(username, password)
        imap.select(mailbox)
        status, response = imap.search(None, 'UNSEEN')
        if status != 'OK':
            try:
                imap.logout()
            except Exception:
                pass
            return results
        ids = response[0].split()
        ids = ids[-limit:]
        for num in ids:
            try:
                res, msg_data = imap.fetch(num, '(RFC822)')
                if res != 'OK':
                    continue
                raw = msg_data[0][1]
                msg = email.message_from_bytes(raw)
                subject = _decode_mime_words(msg.get('Subject', ''))
                from_ = _decode_mime_words(msg.get('From', ''))
                body = ''
                if msg.is_multipart():
                    for part in msg.walk():
                        ctype = part.get_content_type()
                        disp = str(part.get('Content-Disposition'))
                        if ctype == 'text/plain' and 'attachment' not in disp:
                            try:
                                body_bytes = part.get_payload(decode=True)
                                body = body_bytes.decode(part.get_content_charset() or 'utf-8', errors='ignore')
                                break
                            except Exception:
                                continue
                    if not body:
                        for part in msg.walk():
                            if part.get_content_type() == 'text/html':
                                body_bytes = part.get_payload(decode=True)
                                body = body_bytes.decode(part.get_content_charset() or 'utf-8', errors='ignore')
                                break
                else:
                    body_bytes = msg.get_payload(decode=True)
                    if body_bytes:
                        body = body_bytes.decode(msg.get_content_charset() or 'utf-8', errors='ignore')
                    else:
                        body = str(msg.get_payload())
                results.append({'subject': subject, 'from': from_, 'body': body})
                if mark_seen:
                    try:
                        imap.store(num, '+FLAGS', '\\Seen')
                    except Exception:
                        pass
            except Exception:
                continue
        try:
            imap.close()
        except Exception:
            pass
        try:
            imap.logout()
        except Exception:
            pass
    except Exception as e:
        msg = str(e)
